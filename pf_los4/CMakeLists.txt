cmake_minimum_required(VERSION 3.10.2)
project(pf_los4 LANGUAGES CXX)
find_package(MAVSDK 2 REQUIRED)
find_package(Threads REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(pf_los4 src/pf_los4.cpp src/los4_runtime.cpp)
target_compile_features(pf_los4 PRIVATE cxx_std_17)
target_compile_options(pf_los4 PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(pf_los4 PRIVATE MAVSDK::mavsdk Threads::Threads Eigen3::Eigen)
set_target_properties(pf_los4 PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
install(TARGETS pf_los4 DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(los4_core_test tests/los4_core_test.cpp)
  target_compile_features(los4_core_test PRIVATE cxx_std_17)
  target_include_directories(los4_core_test PRIVATE src)
  target_link_libraries(los4_core_test Eigen3::Eigen)
  add_test(NAME los4_core_and_safety COMMAND los4_core_test)
  add_test(NAME pf_los4_help COMMAND pf_los4 --help)
  add_test(NAME pf_los4_invalid_rate COMMAND pf_los4 udp://:14540 -1)
  set_tests_properties(pf_los4_invalid_rate PROPERTIES WILL_FAIL TRUE)
  add_test(NAME pf_los4_invalid_forwarding COMMAND pf_los4 udp://:14540 10 tcp://127.0.0.1:8888)
  set_tests_properties(pf_los4_invalid_forwarding PROPERTIES WILL_FAIL TRUE)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_FOUND)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import pymavlink" RESULT_VARIABLE pymavlink_status OUTPUT_QUIET ERROR_QUIET)
    if(pymavlink_status EQUAL 0)
      add_test(NAME los4_bench_offboard COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bench_offboard.py $<TARGET_FILE:pf_los4>)
      add_test(NAME los4_controller_link COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_controller_link.py $<TARGET_FILE:pf_los4>)
      add_test(NAME pf_los4_forwarding COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_forwarding.py $<TARGET_FILE:pf_los4>)
    endif()
  endif()
endif()
